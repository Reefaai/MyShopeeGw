// ============================================================================
// Cuplikan Kode Tunggal: Model Data (DTO) & Layanan Jaringan Supabase PostgREST
// File: ShopeeApiService.kt & DTO Model ShopiKu (MyShopeeGw)
// ============================================================================

// 1. DTO Model Produk & Keranjang Belanja
data class Product(
    @SerializedName("id") val id: String,
    @SerializedName("name") val name: String,
    @SerializedName("category") val category: String,
    @SerializedName("price") val price: Long,
    @SerializedName("rating") val rating: Double = 5.0
) : Serializable

data class CartItem(
    @SerializedName("id") val id: String,
    @SerializedName("product_id") val productId: String,
    @SerializedName("product_name") val productName: String,
    @SerializedName("price") val price: Long,
    @SerializedName("quantity") var quantity: Int = 1,
    @SerializedName("variant") val variant: String = "",
    @Transient var isSelected: Boolean = true // Local UI state (tidak dikirim ke database Supabase)
) : Serializable

// 2. Interface Retrofit Service Penghubung 7 Alur Transaksi ke Cloud Supabase
interface SupabaseApiService {

    // Alur 1 & 2: Katalog Produk & Pencarian Cepat
    @GET("rest/v1/products?select=*")
    suspend fun getProducts(@Query("name") searchFilter: String? = null): Response<List<Product>>

    // Alur 4: Mengambil & Menambahkan Produk ke Keranjang Belanja
    @GET("rest/v1/cart?select=*")
    suspend fun getCart(): Response<List<CartItem>>

    @Headers("Prefer: return=representation")
    @POST("rest/v1/cart")
    suspend fun addToCart(@Body item: CartItem): Response<List<CartItem>>

    // Alur 5: Mengubah Kuantitas (+/-) & Menghapus Item Keranjang (CRUD)
    @Headers("Prefer: return=representation")
    @PATCH("rest/v1/cart")
    suspend fun updateCartQuantity(@Query("id") idFilter: String, @Body body: Map<String, Int>): Response<List<CartItem>>

    @DELETE("rest/v1/cart")
    suspend fun deleteCartItem(@Query("id") idFilter: String): Response<Void>

    // Alur 6 & 7: Menerbitkan Pesanan Checkout & Verifikasi Status Pembayaran
    @Headers("Prefer: return=representation")
    @POST("rest/v1/orders")
    suspend fun createOrder(@Body order: Map<String, Any>): Response<List<Map<String, Any>>>

    @Headers("Prefer: return=representation")
    @PATCH("rest/v1/orders")
    suspend fun updateOrder(@Query("id") idFilter: String, @Body body: Map<String, Any>): Response<List<Map<String, Any>>>
}
